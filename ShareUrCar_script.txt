-- ShareUrCar DB

DROP TABLE IF EXISTS notifications CASCADE;
DROP TABLE IF EXISTS reviews CASCADE;
DROP TABLE IF EXISTS payments CASCADE;
DROP TABLE IF EXISTS messages CASCADE;
DROP TABLE IF EXISTS group_passengers CASCADE;
DROP TABLE IF EXISTS travel_groups CASCADE;
DROP TABLE IF EXISTS route_passengers CASCADE;
DROP TABLE IF EXISTS routes CASCADE;
DROP TABLE IF EXISTS drivers CASCADE;
DROP TABLE IF EXISTS users CASCADE;

CREATE TABLE users (
                       id_user SERIAL PRIMARY KEY,
                       firstname VARCHAR(100) NOT NULL,
                       lastname VARCHAR(100) NOT NULL,
                       email VARCHAR(150) UNIQUE NOT NULL,
                       password VARCHAR(255) NOT NULL,
                       center VARCHAR(150),
                       profile_photo VARCHAR(255),
                       rating DECIMAL(3,2) DEFAULT 0,
                       phone VARCHAR(20),
                       about_me TEXT,
                       balance DECIMAL(10,2) DEFAULT 0.0,
                       held_balance DECIMAL(10,2) DEFAULT 0.0,
                       created_at DATE DEFAULT CURRENT_DATE
);

CREATE TABLE drivers (
                         car_plate VARCHAR(20) PRIMARY KEY,
                         id_driver INT UNIQUE,
                         car_model VARCHAR(100),
                         max_seats INT NOT NULL,
                         car_color VARCHAR(50),
                         CONSTRAINT fk_driver_user FOREIGN KEY (id_driver) REFERENCES users(id_user) ON DELETE CASCADE
);

CREATE TABLE routes (
                        id_route SERIAL PRIMARY KEY,
                        id_driver INT,
                        origin VARCHAR(150) NOT NULL,
                        destination VARCHAR(150) NOT NULL,
                        departure_time TIME NOT NULL,
                        arrival_time TIME,
                        return_time TIME,
                        days_of_week VARCHAR(50),
                        frequency VARCHAR(50),
                        available_seats INT NOT NULL,
                        status VARCHAR(20) DEFAULT 'PENDING',
                        driver_confirmed BOOLEAN DEFAULT FALSE,
                        allow_round_trip BOOLEAN DEFAULT FALSE,
                        pref_no_talk BOOLEAN DEFAULT FALSE,
                        pref_luggage BOOLEAN DEFAULT FALSE,
                        pref_music BOOLEAN DEFAULT FALSE,
                        pref_smoke BOOLEAN DEFAULT FALSE,
                        origin_lat DECIMAL(10,7),
                        origin_lng DECIMAL(10,7),
                        destination_lat DECIMAL(10,7),
                        destination_lng DECIMAL(10,7),
                        travel_date DATE,
                        start_date DATE,
                        end_date DATE,
                        series_id VARCHAR(100),
                        CONSTRAINT fk_route_driver FOREIGN KEY (id_driver) REFERENCES users(id_user) ON DELETE CASCADE
);

CREATE TABLE route_passengers (
                                  id_route INT NOT NULL,
                                  id_user INT NOT NULL,
                                  PRIMARY KEY (id_route, id_user),
                                  CONSTRAINT fk_rp_route FOREIGN KEY (id_route) REFERENCES routes(id_route) ON DELETE CASCADE,
                                  CONSTRAINT fk_rp_user FOREIGN KEY (id_user) REFERENCES users(id_user) ON DELETE CASCADE
);

CREATE TABLE travel_groups (
                               id_group SERIAL PRIMARY KEY,
                               id_route INT,
                               id_driver INT,
                               status VARCHAR(20) DEFAULT 'ACTIVE',
                               travel_date DATE,
                               travel_time TIME,
                               series_id VARCHAR(100),
                               CONSTRAINT fk_group_route FOREIGN KEY (id_route) REFERENCES routes(id_route) ON DELETE CASCADE,
                               CONSTRAINT fk_group_driver FOREIGN KEY (id_driver) REFERENCES users(id_user) ON DELETE CASCADE
);

CREATE TABLE group_passengers (
                                  id_group_passenger SERIAL PRIMARY KEY,
                                  id_group INT,
                                  id_user INT,
                                  state VARCHAR(20) DEFAULT 'ACTIVE',
                                  confirmed BOOLEAN DEFAULT FALSE,
                                  CONSTRAINT fk_gp_group FOREIGN KEY (id_group) REFERENCES travel_groups(id_group) ON DELETE CASCADE,
                                  CONSTRAINT fk_gp_user FOREIGN KEY (id_user) REFERENCES users(id_user) ON DELETE CASCADE,
                                  CONSTRAINT unique_group_user UNIQUE (id_group, id_user)
);

CREATE TABLE messages (
                          id_message SERIAL PRIMARY KEY,
                          id_group INT,
                          id_user INT,
                          text TEXT NOT NULL,
                          sent_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                          CONSTRAINT fk_msg_group FOREIGN KEY (id_group) REFERENCES travel_groups(id_group) ON DELETE CASCADE,
                          CONSTRAINT fk_msg_user FOREIGN KEY (id_user) REFERENCES users(id_user) ON DELETE CASCADE
);

CREATE TABLE payments (
                          id_payment SERIAL PRIMARY KEY,
                          id_group INT,
                          id_user INT,
                          amount DECIMAL(8,2) NOT NULL,
                          payment_status VARCHAR(20) DEFAULT 'PENDING',
                          payment_type VARCHAR(30),
                          trip_type VARCHAR(20),
                          created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                          CONSTRAINT fk_payment_group FOREIGN KEY (id_group) REFERENCES travel_groups(id_group) ON DELETE CASCADE,
                          CONSTRAINT fk_payment_user FOREIGN KEY (id_user) REFERENCES users(id_user) ON DELETE CASCADE
);

CREATE TABLE reviews (
                         id_review SERIAL PRIMARY KEY,
                         id_reviewer INT NOT NULL,
                         id_reviewed INT NOT NULL,
                         stars INT CHECK (stars >= 1 AND stars <= 5),
                         comment TEXT,
                         created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                         CONSTRAINT fk_reviewer FOREIGN KEY (id_reviewer) REFERENCES users(id_user) ON DELETE CASCADE,
                         CONSTRAINT fk_reviewed FOREIGN KEY (id_reviewed) REFERENCES users(id_user) ON DELETE CASCADE
);

CREATE TABLE notifications (
                               id_notification SERIAL PRIMARY KEY,
                               id_user INT NOT NULL,
                               title VARCHAR(150) NOT NULL,
                               body TEXT,
                               is_read BOOLEAN DEFAULT FALSE,
                               created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                               CONSTRAINT fk_notif_user FOREIGN KEY (id_user) REFERENCES users(id_user) ON DELETE CASCADE
);
